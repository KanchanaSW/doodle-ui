## Usage

```tsx
import { Blockquote, Heading, Highlight, InlineCode, Text } from "doodleui-react";

<Heading level={1} accent="underline">Notes</Heading>
<Text variant="lead">A quick intro.</Text>
<Text>Marked <Highlight>important</Highlight> and <InlineCode>npm i</InlineCode>.</Text>
<Blockquote>Sketch borders optional.</Blockquote>
```
