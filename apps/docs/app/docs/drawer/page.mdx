# Drawer

Bottom drawer with sketch pull handle.

<ComponentPlayground name="drawer" />

## CLI install

```bash
npx doodleui-react add drawer
```

## Usage

```tsx
import { Drawer, DrawerTrigger, DrawerContent, DrawerHeader, DrawerTitle, Button } from "doodleui-react";

<Drawer>\n  <DrawerTrigger asChild>\n    <Button>Open</Button>\n  </DrawerTrigger>\n  <DrawerContent>\n    <DrawerHeader>\n      <DrawerTitle>Title</DrawerTitle>\n    </DrawerHeader>\n  </DrawerContent>\n</Drawer>
```
