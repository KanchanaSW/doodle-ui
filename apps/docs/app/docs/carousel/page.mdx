# Carousel

Embla-powered slides with circular sketch arrows outside the viewport, matching the usual carousel layout.

<ComponentPlayground name="carousel" />

<ComponentCliInstallTabs name="carousel" />

## Usage

```tsx
import {
  Card,
  Carousel,
  CarouselContent,
  CarouselItem,
  CarouselNext,
  CarouselPrevious,
} from "doodleui-react";

<div style={{ maxWidth: 280, margin: "0 auto" }}>
  <Carousel>
    <CarouselContent>
      <CarouselItem>
        <Card style={{ aspectRatio: "1 / 1" }}>1</Card>
      </CarouselItem>
      <CarouselItem>
        <Card style={{ aspectRatio: "1 / 1" }}>2</Card>
      </CarouselItem>
    </CarouselContent>
    <CarouselPrevious />
    <CarouselNext />
  </Carousel>
</div>
```
